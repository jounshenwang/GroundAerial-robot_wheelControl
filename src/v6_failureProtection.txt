// ===================== ESP32-S3 陆空两栖机器人 串级PID+安全保护版 =====================
#include <Arduino.h>
#include <driver/ledc.h>

// [1. 引脚定义]
#define PX4_CH3_PIN 10   // 油门 -> 控制位置增量速率
#define PX4_CH4_PIN 9    // 偏航 -> 控制差速增量
#define PX4_CH9_PIN 46    // 模式切换
#define STBY_PIN 4
#define PWMA_PIN 5
#define AIN1_PIN 7
#define AIN2_PIN 6
#define PWMD_PIN 38
#define DIN1_PIN 37
#define DIN2_PIN 39
#define ENC_L_A 15
#define ENC_L_B 16
 #define ENC_R_A 36
#define ENC_R_B 35

// [2. 串级 PID 结构体与参数]
struct CascadePID {
    // 位置外环：决定“要走多远”
    float kp_p = 1.2, ki_p = 0.01;
    float p_integral = 0;
    
    // 速度内环：决定“转得稳不稳”
    float kp_v = 8.5, ki_v = 1.2, kd_v = 0.5;
    float v_integral = 0, v_last_err = 0;
};
CascadePID pidL, pidR;

// [3. 全局变量与常量]
volatile long count_l = 0, count_r = 0; 
long targetPosL = 0, targetPosR = 0;    
volatile uint32_t ch3=1500, ch4=1500, ch9=1000;

#define MAX_V_TARGET 150    // 外环输出给内环的最大速度指令
#define ERROR_LIMIT 800     // 【保护】反暴冲限幅：目标领先实际位置的最大脉冲数
#define I_V_LIMIT 80        // 【保护】内环积分限幅：防止卡死时电机烧毁

// [4. 中断回调：编码器与接收机]
void IRAM_ATTR encL() { if(digitalRead(ENC_L_A) != digitalRead(ENC_L_B)) count_l++; else count_l--; }
void IRAM_ATTR encR() { if(digitalRead(ENC_R_A) != digitalRead(ENC_R_B)) count_r++; else count_r--; }
void readCh3() { static uint32_t t=0; if(digitalRead(PX4_CH3_PIN)) t=micros(); else ch3=constrain(micros()-t,1000,2000); }
void readCh4() { static uint32_t t=0; if(digitalRead(PX4_CH4_PIN)) t=micros(); else ch4=constrain(micros()-t,1000,2000); }
void readCh9() { static uint32_t t=0; if(digitalRead(PX4_CH9_PIN)) t=micros(); else ch9=constrain(micros()-t,1000,2000); }

// [5. 串级 PID 核心算法]
int computeCascade(CascadePID &p, long target, long actualPos, int actualSpeed) {
    // 外环：位置控制
    float p_err = target - actualPos;
    p.p_integral = constrain(p.p_integral + p_err, -50, 50);
    float v_target = p.kp_p * p_err + p.ki_p * p.p_integral;
    v_target = constrain(v_target, -MAX_V_TARGET, MAX_V_TARGET); 

    // 内环：速度控制
    float v_err = v_target - actualSpeed;
    p.v_integral = constrain(p.v_integral + v_err, -I_V_LIMIT, I_V_LIMIT); // 【保护：限流】
    float output = p.kp_v * v_err + p.ki_v * p.v_integral + p.kd_v * (v_err - p.v_last_err);
    p.v_last_err = v_err;

    return (int)constrain(output, -255, 255);
}

// [6. 电机底层驱动逻辑]
void setMotor(int out, int p1, int p2, int channel) {
    if (out > 5) { // 正向运动
        digitalWrite(p1, HIGH); digitalWrite(p2, LOW);
    } else if (out < -5) { // 反向运动
        digitalWrite(p1, LOW); digitalWrite(p2, HIGH);
    } else { // 【保护：驻车】IN1=IN2=HIGH 产生电子制动
        digitalWrite(p1, HIGH); digitalWrite(p2, HIGH);
    }
    ledcWrite(channel, constrain(abs(out), 0, 255));
}

void setup() {
    Serial.begin(115200);
    pinMode(STBY_PIN, OUTPUT); digitalWrite(STBY_PIN, HIGH);
    pinMode(AIN1_PIN, OUTPUT); pinMode(AIN2_PIN, OUTPUT);
    pinMode(DIN1_PIN, OUTPUT); pinMode(DIN2_PIN, OUTPUT);
    
    ledcSetup(0, 5000, 8); ledcSetup(1, 5000, 8);
    ledcAttachPin(PWMA_PIN, 0); ledcAttachPin(PWMD_PIN, 1);

    attachInterrupt(digitalPinToInterrupt(PX4_CH3_PIN), readCh3, CHANGE);
    attachInterrupt(digitalPinToInterrupt(PX4_CH4_PIN), readCh4, CHANGE);
    attachInterrupt(digitalPinToInterrupt(PX4_CH9_PIN), readCh9, CHANGE);
    attachInterrupt(digitalPinToInterrupt(ENC_L_A), encL, CHANGE);
    attachInterrupt(digitalPinToInterrupt(ENC_R_A), encR, CHANGE);
}

void loop() {
    // 强制 20ms 采样周期 (50Hz)
    static uint32_t lastSampleTime = 0;
    if (millis() - lastSampleTime < 20) return;
    lastSampleTime = millis();
    
    // A. 速度计算
    static long last_count_l = 0, last_count_r = 0;
    int act_speed_l = count_l - last_count_l;
    int act_speed_r = count_r - last_count_r;
    last_count_l = count_l; last_count_r = count_r;

    // B. 地面控制逻辑
    int outL = 0, outR = 0;
    if (ch9 > 1700) { 
        int move_step = 0;
        int steer_step = 0;

        // 【保护：链路安全检测】
        // 只有当信号在正常脉宽范围内(900~2100us)才允许移动，否则 move_step 为 0（驻车）
        if (ch3 > 900 && ch3 < 2100 && ch4 > 900 && ch4 < 2100) {
            if (ch3 > 1580 || ch3 < 1420) move_step = map(ch3, 1000, 2000, -60, 60); 
            if (ch4 > 1550 || ch4 < 1450) steer_step = map(ch4, 1000, 2000, -30, 30);
        }

        // 应用位置增量
        targetPosL += (move_step + steer_step);
        targetPosR += (move_step - steer_step);

        // 【保护：反暴冲】限制目标超前量
        targetPosL = constrain(targetPosL, count_l - ERROR_LIMIT, count_l + ERROR_LIMIT);
        targetPosR = constrain(targetPosR, count_r - ERROR_LIMIT, count_r + ERROR_LIMIT);

        // 串级 PID 计算与执行
        outL = computeCascade(pidL, targetPosL, count_l, act_speed_l);
        outR = computeCascade(pidR, targetPosR, count_r, act_speed_r);

        setMotor(outL, AIN1_PIN, AIN2_PIN, 0);
        setMotor(outR, DIN1_PIN, DIN2_PIN, 1);
    } 
    // C. 飞行模式/安全复位
    else {
        // 【保护：初始位错清零】实时同步位置，确保切回地面模式时无跳变
        targetPosL = count_l; targetPosR = count_r;
        setMotor(0, AIN1_PIN, AIN2_PIN, 0);
        setMotor(0, DIN1_PIN, DIN2_PIN, 1);
        
        // 清空积分项，防止模式切换后的累积误差
        pidL.p_integral = 0; pidL.v_integral = 0;
        pidR.p_integral = 0; pidR.v_integral = 0;
    }

    // D. 串口监控
    static uint32_t t = 0;
    if (millis() - t > 100) {
        Serial.printf("L: Target:%ld Actual:%ld Out:%d | R: Target:%ld Actual:%ld Out:%d\n", targetPosL, count_l, outL, targetPosR, count_r, outR);
        t = millis();
    }
}
