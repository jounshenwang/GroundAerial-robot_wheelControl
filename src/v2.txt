// ===================== ESP32-S3 优化修正版 =====================
#include <Arduino.h>
#include <driver/ledc.h>

// [硬件引脚定义保持不变...]
#define PX4_AUX1_PIN    10 
#define PX4_AUX2_PIN     9 
#define PX4_CH9_PIN      8 
#define ENC_L_A         15
#define ENC_L_B         16
#define ENC_R_A         38
#define ENC_R_B         39
#define STBY_PIN         7
#define PWMA_PIN         4
#define AIN1_PIN         6
#define AIN2_PIN         5
#define PWMD_PIN        37
#define DIN1_PIN        35
#define DIN2_PIN        36

// [参数定义]
#define DEAD_BAND_MIN    1420 
#define DEAD_BAND_MAX    1580
#define GROUND_MODE      1700
#define MAX_SPEED        180

// ===================== 【修改1：引入PID结构体，方便一键复位】 =====================
struct PID_Data {
    float integral = 0;
    float last_err = 0;
};
PID_Data pidL, pidR; // 定义左、右电机独立的PID状态

volatile uint32_t ch3=1500, ch4=1500, ch5=1000;
bool is_ground = false;
volatile long count_l=0, count_r=0;

// [PWM读取函数保持不变]
void readCh3() { static uint32_t t=0; if(digitalRead(PX4_AUX1_PIN)) t=micros(); else ch3=constrain(micros()-t,1000,2000); }
void readCh4() { static uint32_t t=0; if(digitalRead(PX4_AUX2_PIN)) t=micros(); else ch4=constrain(micros()-t,1000,2000); }
void readCh5() { static uint32_t t=0; if(digitalRead(PX4_CH9_PIN)) t=micros(); else ch5=constrain(micros()-t,1000,2000); }

void IRAM_ATTR encL() { if(digitalRead(ENC_L_A) != digitalRead(ENC_L_B)) count_l++; else count_l--; }
void IRAM_ATTR encR() { if(digitalRead(ENC_R_A) != digitalRead(ENC_R_B)) count_r++; else count_r--; }

// ===================== 【修改2：重写 motorStop，强制清空PID记忆】 =====================
void motorStop(){
  digitalWrite(STBY_PIN, LOW);
  digitalWrite(AIN1_PIN, LOW); digitalWrite(AIN2_PIN, LOW);
  digitalWrite(DIN1_PIN, LOW); digitalWrite(DIN2_PIN, LOW);
  ledcWrite(0, 0); ledcWrite(1, 0);
  
  // 关键：切换模式或停止时，必须把积分项清零，否则下次启动会暴冲
  pidL.integral = 0; pidL.last_err = 0;
  pidR.integral = 0; pidR.last_err = 0;
}

void motorLeft(int dir, uint8_t pwm){
  digitalWrite(STBY_PIN, HIGH);
  if(dir>0){ digitalWrite(AIN1_PIN,HIGH); digitalWrite(AIN2_PIN,LOW); }
  else{ digitalWrite(AIN1_PIN,LOW); digitalWrite(AIN2_PIN,HIGH); }
  ledcWrite(0,pwm);
}

void motorRight(int dir, uint8_t pwm){
  if(dir>0){ digitalWrite(DIN1_PIN,HIGH); digitalWrite(DIN2_PIN,LOW); }
  else{ digitalWrite(DIN1_PIN,LOW); digitalWrite(DIN2_PIN,HIGH); }
  ledcWrite(1,pwm);
}

// ===================== 【修改3：优化PID算法，增加指针传递】 =====================
int runPID(PID_Data &p, int target, int actual, float Kp, float Ki, float Kd){
  if(target == 0) return 0;
  
  float err = target - actual;
  p.integral += err;
  p.integral = constrain(p.integral, -150, 150); // 限制积分限幅
  
  float out = Kp * err + Ki * p.integral + Kd * (err - p.last_err);
  p.last_err = err;
  
  return constrain((int)out, 0, 255);
}

void setup() {
  Serial.begin(115200);
  pinMode(STBY_PIN,OUTPUT);
  pinMode(AIN1_PIN,OUTPUT); pinMode(AIN2_PIN,OUTPUT);
  pinMode(DIN1_PIN,OUTPUT); pinMode(DIN2_PIN,OUTPUT);

  ledcSetup(0,5000,8); // 频率提高到5kHz，电机运行更平滑
  ledcSetup(1,5000,8);
  ledcAttachPin(PWMA_PIN,0); ledcAttachPin(PWMD_PIN,1);

  attachInterrupt(digitalPinToInterrupt(PX4_AUX1_PIN),readCh3,CHANGE);
  attachInterrupt(digitalPinToInterrupt(PX4_AUX2_PIN),readCh4,CHANGE);
  attachInterrupt(digitalPinToInterrupt(PX4_CH9_PIN),readCh5,CHANGE);

  pinMode(ENC_L_A,INPUT); pinMode(ENC_L_B,INPUT);
  pinMode(ENC_R_A,INPUT); pinMode(ENC_R_B,INPUT);
  attachInterrupt(digitalPinToInterrupt(ENC_L_A),encL,CHANGE);
  attachInterrupt(digitalPinToInterrupt(ENC_R_A),encR,CHANGE);

  motorStop();
}

// ===================== 【修改4：优化主循环逻辑，增强互锁】 =====================
void loop() {
  int actL = abs(count_l); count_l = 0;
  int actR = abs(count_r); count_r = 0;
  is_ground = (ch5 > GROUND_MODE);

  // 1. 严格互锁判断
  // 只有在地面模式 且 油门离开中位 且 信号有效 时才执行逻辑
  if (!is_ground || (ch3 > DEAD_BAND_MIN && ch3 < DEAD_BAND_MAX) || ch3 < 900) {
    motorStop(); // 不满足条件时，电机物理停止+PID状态复位
  } 
  else {
    // 2. 映射油门（确保从0开始）
    int base = 0;
    if(ch3 >= DEAD_BAND_MAX) base = map(ch3, DEAD_BAND_MAX, 2000, 0, MAX_SPEED);
    else if(ch3 <= DEAD_BAND_MIN) base = map(ch3, DEAD_BAND_MIN, 1000, 0, -MAX_SPEED);

    // 3. 映射偏航（Yaw），加入小死区防止微小自转
    int steer = 0;
    if(ch4 > 1550 || ch4 < 1450) {
      steer = map(ch4, 1000, 2000, -60, 60);
    }

    int targetL = constrain(base + steer, -MAX_SPEED, MAX_SPEED);
    int targetR = constrain(base - steer, -MAX_SPEED, MAX_SPEED);

    // 4. 调用带状态记忆的PID
    int pwm_l = runPID(pidL, abs(targetL), actL, 7.5, 1.0, 0.2);
    int pwm_r = runPID(pidR, abs(targetR), actR, 7.5, 1.0, 0.2);
    
    motorLeft(targetL > 0 ? 1 : -1, pwm_l);
    motorRight(targetR > 0 ? 1 : -1, pwm_r);
  }

  // 串口监控，方便QGC与遥控器对齐
  static uint32_t pTick = 0;
  if(millis() - pTick > 100) {
    Serial.printf("Mode:%s Thr:%d L:%d R:%d\n", is_ground?"GND":"AIR", ch3, actL, actR);
    pTick = millis();
  }
  delay(20); 
}
