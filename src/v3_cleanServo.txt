// ===================== ESP32-S3 终极整合版 =====================
#include <Arduino.h>
#include <driver/ledc.h>

// [1. 接收机输入引脚]
#define PX4_CH3_PIN     10  // 油门 (原 AUX1)
#define PX4_CH4_PIN      9  // 偏航 (原 AUX2)
#define PX4_CH9_PIN      8  // 陆空模式切换 (原 CH5 逻辑)
#define PX4_CH10_PIN     3  // 额外电机独立开关

// [2. 行驶电机引脚 (A路 & D路)]
#define STBY_PIN         7  
#define PWMA_PIN         4  
#define AIN1_PIN         6  
#define AIN2_PIN         5  
#define PWMD_PIN        37  
#define DIN1_PIN        35  
#define DIN2_PIN        36  

// [3. 额外电机引脚 (C路)]
#define CIN1_PIN        40  
#define CIN2_PIN        41  
#define PWMC_PIN        42  

// [4. 编码器引脚]
#define ENC_L_A         15
#define ENC_L_B         16
#define ENC_R_A         38
#define ENC_R_B         39

// [5. 参数定义]
#define DEAD_BAND_MIN    1420 
#define DEAD_BAND_MAX    1580
#define GROUND_MODE_VAL  1700 // CH9 阈值
#define EXTRA_ON_VAL     1500 // CH10 阈值
#define MAX_SPEED        180

struct PID_Data {
    float integral = 0;
    float last_err = 0;
};
PID_Data pidL, pidR;

volatile uint32_t ch3=1500, ch4=1500, ch9=1000, ch10=1000;
volatile long count_l=0, count_r=0;

// [PWM信号读取]
void readCh3() { static uint32_t t=0; if(digitalRead(PX4_CH3_PIN)) t=micros(); else ch3=constrain(micros()-t,1000,2000); }
void readCh4() { static uint32_t t=0; if(digitalRead(PX4_CH4_PIN)) t=micros(); else ch4=constrain(micros()-t,1000,2000); }
void readCh9() { static uint32_t t=0; if(digitalRead(PX4_CH9_PIN)) t=micros(); else ch9=constrain(micros()-t,1000,2000); }
void readCh10(){ static uint32_t t=0; if(digitalRead(PX4_CH10_PIN)) t=micros(); else ch10=constrain(micros()-t,1000,2000); }

void IRAM_ATTR encL() { if(digitalRead(ENC_L_A) != digitalRead(ENC_L_B)) count_l++; else count_l--; }
void IRAM_ATTR encR() { if(digitalRead(ENC_R_A) != digitalRead(ENC_R_B)) count_r++; else count_r--; }

// 停止行驶电机（不关STBY，不影响C路）
void driveMotorsStop(){
    digitalWrite(AIN1_PIN, LOW); digitalWrite(AIN2_PIN, LOW);
    digitalWrite(DIN1_PIN, LOW); digitalWrite(DIN2_PIN, LOW);
    ledcWrite(0, 0); ledcWrite(1, 0);
    pidL.integral = 0; pidL.last_err = 0;
    pidR.integral = 0; pidR.last_err = 0;
}

// PID 算法
int runPID(PID_Data &p, int target, int actual, float Kp, float Ki, float Kd){
    if(target == 0) return 0;
    float err = target - actual;
    p.integral += err;
    p.integral = constrain(p.integral, -150, 150);
    float out = Kp * err + Ki * p.integral + Kd * (err - p.last_err);
    p.last_err = err;
    return constrain((int)out, 0, 255);
}

void setup() {
    Serial.begin(115200);
    
    // 初始化所有输出引脚
    pinMode(STBY_PIN, OUTPUT);
    pinMode(AIN1_PIN, OUTPUT); pinMode(AIN2_PIN, OUTPUT);
    pinMode(DIN1_PIN, OUTPUT); pinMode(DIN2_PIN, OUTPUT);
    pinMode(CIN1_PIN, OUTPUT); pinMode(CIN2_PIN, OUTPUT);
    pinMode(PWMC_PIN, OUTPUT);

    // 行驶电机PWM
    ledcSetup(0, 5000, 8); 
    ledcSetup(1, 5000, 8);
    ledcAttachPin(PWMA_PIN, 0); 
    ledcAttachPin(PWMD_PIN, 1);

    // 接收机中断
    attachInterrupt(digitalPinToInterrupt(PX4_CH3_PIN),  readCh3,  CHANGE);
    attachInterrupt(digitalPinToInterrupt(PX4_CH4_PIN),  readCh4,  CHANGE);
    attachInterrupt(digitalPinToInterrupt(PX4_CH9_PIN),  readCh9,  CHANGE);
    attachInterrupt(digitalPinToInterrupt(PX4_CH10_PIN), readCh10, CHANGE);

    // 编码器中断
    pinMode(ENC_L_A, INPUT); pinMode(ENC_L_B, INPUT);
    pinMode(ENC_R_A, INPUT); pinMode(ENC_R_B, INPUT);
    attachInterrupt(digitalPinToInterrupt(ENC_L_A), encL, CHANGE);
    attachInterrupt(digitalPinToInterrupt(ENC_R_A), encR, CHANGE);

    digitalWrite(STBY_PIN, HIGH); // 全局使能开启
}

void loop() {
    int actL = abs(count_l); count_l = 0;
    int actR = abs(count_r); count_r = 0;

    // --- 1. CH10 逻辑：附件电机独立开关 (C路) ---
    if (ch10 > EXTRA_ON_VAL) {
        digitalWrite(CIN1_PIN, HIGH);
        digitalWrite(CIN2_PIN, LOW);
        digitalWrite(PWMC_PIN, HIGH);
    } else {
        digitalWrite(CIN1_PIN, LOW);
        digitalWrite(CIN2_PIN, LOW);
        digitalWrite(PWMC_PIN, LOW);
    }

    // --- 2. CH9 逻辑：陆空模式与行驶 (A路 & D路) ---
    bool is_ground = (ch9 > GROUND_MODE_VAL);

    if (!is_ground || (ch3 > DEAD_BAND_MIN && ch3 < DEAD_BAND_MAX) || ch3 < 900) {
        driveMotorsStop(); 
    } 
    else {
        // 计算 base 和 steer
        int base = 0;
        if(ch3 >= DEAD_BAND_MAX) base = map(ch3, DEAD_BAND_MAX, 2000, 0, MAX_SPEED);
        else if(ch3 <= DEAD_BAND_MIN) base = map(ch3, DEAD_BAND_MIN, 1000, 0, -MAX_SPEED);

        int steer = (ch4 > 1550 || ch4 < 1450) ? map(ch4, 1000, 2000, -60, 60) : 0;

        int targetL = constrain(base + steer, -MAX_SPEED, MAX_SPEED);
        int targetR = constrain(base - steer, -MAX_SPEED, MAX_SPEED);

        int pwm_l = runPID(pidL, abs(targetL), actL, 7.5, 1.0, 0.2);
        int pwm_r = runPID(pidR, abs(targetR), actR, 7.5, 1.0, 0.2);
        
        // 物理输出
        if(targetL > 0){ digitalWrite(AIN1_PIN, HIGH); digitalWrite(AIN2_PIN, LOW); }
        else { digitalWrite(AIN1_PIN, LOW); digitalWrite(AIN2_PIN, HIGH); }
        ledcWrite(0, pwm_l);

        if(targetR > 0){ digitalWrite(DIN1_PIN, HIGH); digitalWrite(DIN2_PIN, LOW); }
        else { digitalWrite(DIN1_PIN, LOW); digitalWrite(DIN2_PIN, HIGH); }
        ledcWrite(1, pwm_r);
    }

    // 监控
    static uint32_t pTick = 0;
    if(millis() - pTick > 100) {
        Serial.printf("Mode_CH9:%d Extra_CH10:%d L:%d R:%d\n", ch9, ch10, actL, actR);
        pTick = millis();
    }
    delay(20); 
}
